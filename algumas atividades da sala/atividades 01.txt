#include <stdio.h>
void dobra_valor(int x)  { x *= 2; }
void dobra_ref(int *x)   { *x *= 2; }
void print_array(int *v, int n) {
    for (int i = 0; i < n; i++) printf("%d\n", v[i]);
}
int main(void) {
    int array[] = {1, 2, 3, 4, 5};
    print_array((int *)array, 5);
    for (int i = 0; i < 5; i++) dobra_valor(array[i]);
    print_array((int *)array, 5);
    for (int i = 0; i < 5; i++) dobra_ref(&array[i]);
    print_array((int *)array, 5);
}
