#include <stdio.h>
 
#define PI            3.14159
#define QUADRADO(x)   ((x)*(x))
#define MAX(a,b)      ((a)>(b)?(a):(b))
 
#define DEBUG 1
#ifdef DEBUG
    #define LOG(msg)  printf("[%s:%d] %s\n", __FILE__, __LINE__, msg)
#else
    #define LOG(msg)
#endif
 
int main(void) {
    double r = 5.0;
    printf("Área r=5: %.2f\n", PI * QUADRADO(r));
 
    int a = 7, b = 3;
    printf("MAX(%d,%d) = %d\n", a, b, MAX(a,b));
 
    LOG("iniciando cálculo");   /* só aparece se DEBUG estiver definido */
 
    printf("Compilado em %s às %s\n", __DATE__, __TIME__);
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * As macros são substituições realizadas pelo pré-processador antes da compilação.
 * QUADRADO(r) calcula r × r, então a área aproximada é 3.14159 × 25 = 78.54. MAX(a,b) escolhe
 * o maior e retorna 7. Como DEBUG está definido, LOG imprime o arquivo, a linha e a mensagem.
 * __DATE__ e __TIME__ mostram a data e a hora de compilação, portanto variam conforme quando
 * o programa é compilado.
 */
