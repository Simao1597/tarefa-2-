#include <stdio.h>
 
int mult(int a, int b) {
    return a * b;
}
 
int soma_mult(int a, int b, int c) {
    return a + mult(b, c);
}
 
int main(void) {
    int resultado = soma_mult(1, 5, 3);
    printf("%d\n", resultado);   /* 16 */
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * O programa começa em main e chama soma_mult(1, 5, 3). Dentro de soma_mult, a expressão
 * mult(b, c) chama mult(5, 3), que devolve 15. Depois soma_mult soma esse resultado ao valor
 * a, que é 1, e devolve 16. main guarda o retorno em resultado e printf mostra 16. No Python
 * Tutor, cada chamada cria um quadro temporário na pilha; o quadro desaparece quando a função
 * termina.
 */
