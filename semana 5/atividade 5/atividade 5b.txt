#include <stdio.h>
 
int main(void) {
    for (int i = 0; i < 10; i++) {
        if (i % 2 == 0) continue;
        if (i == 7)     break;
        printf("%d\n", i);
    }
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * O for percorre i de 0 até 9. Quando i é par, continue pula o restante da repetição. Assim,
 * só os ímpares chegam ao printf. Ao chegar em i = 7, break encerra o laço antes de imprimir
 * 7. A saída é 1, 3 e 5, cada um em uma linha.
 */
