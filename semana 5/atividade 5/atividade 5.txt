#include <stdio.h>
 
int main(void) {
    int nota = 72;
    if      (nota >= 90) printf("A\n");
    else if (nota >= 80) printf("B\n");
    else if (nota >= 70) printf("C\n");
    else                 printf("Reprovado\n");
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * A variável nota vale 72. O programa testa as condições em ordem: não é maior ou igual a 90,
 * nem a 80, mas é maior ou igual a 70. Por isso executa somente a terceira alternativa e
 * imprime C. Quando uma condição de uma cadeia if/else if é verdadeira, as alternativas
 * seguintes são ignoradas.
 */
