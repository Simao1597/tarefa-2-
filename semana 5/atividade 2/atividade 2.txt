#include <stdio.h>
 
int main(void) {
    printf("char:      %zu byte(s)\n", sizeof(char));
    printf("short:     %zu byte(s)\n", sizeof(short));
    printf("int:       %zu byte(s)\n", sizeof(int));
    printf("long:      %zu byte(s)\n", sizeof(long));
    printf("long long: %zu byte(s)\n", sizeof(long long));
    printf("float:     %zu byte(s)\n", sizeof(float));
    printf("double:    %zu byte(s)\n", sizeof(double));
 
    /* Promoção e truncamento */
    printf("\n7 / 2   = %d\n",   7 / 2);       /* 3  */
    printf("7.0 / 2 = %.1f\n",  7.0 / 2);      /* 3.5 */
    printf("(int)3.9= %d\n",    (int)3.9);      /* 3  */
 
    unsigned int u = 0;
    u--;  /* wrap-around: vira UINT_MAX */
    printf("0u - 1  = %u\n", u);
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * sizeof informa quantos bytes cada tipo ocupa nessa implementação. Os tamanhos de long e de
 * alguns outros tipos podem variar conforme o sistema, portanto não se deve assumir que todas
 * as máquinas mostrarão exatamente os mesmos valores. A divisão 7 / 2 usa inteiros e resulta
 * em 3; 7.0 / 2 resulta em 3.5 porque um operando é decimal; (int)3.9 descarta a parte
 * fracionária e resulta em 3. Como u é unsigned, decrementar zero faz o valor dar a volta
 * para o maior valor representável por unsigned int.
 */
