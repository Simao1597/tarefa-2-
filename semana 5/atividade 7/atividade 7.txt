#include <stdio.h>
#include <string.h>
 
int main(void) {
    int v[5] = {10, 20, 30, 40, 50};
    int soma = 0;
    for (int i = 0; i < 5; i++) soma += v[i];
    printf("soma = %d\n", soma);
 
    /* cópia de array com memcpy */
    int w[5];
    memcpy(w, v, sizeof(v));
    w[0] = 99;
    printf("v[0]=%d  w[0]=%d\n", v[0], w[0]);   /* 10 e 99 — independentes */
 
    /* matriz */
    int m[2][3] = {{1, 2, 3}, {4, 5, 6}};
    printf("m[1][2] = %d\n", m[1][2]);   /* 6 */
 
    /* row-major: elementos consecutivos na memória */
    int *p = &m[0][0];
    for (int i = 0; i < 6; i++) printf("%d ", p[i]);   /* 1 2 3 4 5 6 */
    printf("\n");
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * O vetor v contém 10, 20, 30, 40 e 50; o laço soma os cinco valores e imprime 150. memcpy
 * copia os bytes de v para w, criando uma cópia independente; por isso mudar w[0] para 99 não
 * muda v[0], que continua 10. Na matriz, m[1][2] é a linha de índice 1 e coluna de índice 2,
 * cujo valor é 6. A leitura sequencial da memória imprime 1 2 3 4 5 6.
 */
