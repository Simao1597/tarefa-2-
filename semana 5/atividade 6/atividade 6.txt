#include <stdio.h>
 
int fatorial(int n) {
    if (n <= 1) return 1;
    return n * fatorial(n - 1);
}
 
void dobra(int *x) { *x *= 2; }
 
int main(void) {
    printf("5! = %d\n", fatorial(5));
 
    int v = 7;
    dobra(&v);
    printf("dobro de 7 = %d\n", v);
 
    /* ponteiro para função */
    int (*fn)(int) = fatorial;
    printf("fn(4) = %d\n", fn(4));   /* 24 */
    return 0;
}

/* EXPLICAÇÃO DO FUNCIONAMENTO:
 * fatorial calcula 5! recursivamente: 5 × 4 × 3 × 2 × 1 = 120. Cada chamada usa um n menor
 * até atingir o caso-base n <= 1. A função dobra recebe o endereço de v; ao alterar *x,
 * altera a própria variável v, de 7 para 14. Por fim, fn guarda o endereço da função fatorial
 * e fn(4) calcula 4! = 24.
 */
